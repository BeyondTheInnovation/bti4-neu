## Installation

```bash
bunx --bun shadcn@latest add https://neu.beyondtheinnovation.com/r/neu-hover-card.json
```

## Usage

```tsx
import {
  HoverCard,
  HoverCardContent,
  HoverCardTrigger,
} from "@/components/ui/neu/hover-card"
```

```tsx
<HoverCard>
  <HoverCardTrigger>Hover</HoverCardTrigger>
  <HoverCardContent>The React Framework, created and maintained by @vercel.</HoverCardContent>
</HoverCard>
```
