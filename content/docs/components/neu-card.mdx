## Installation

```bash
bunx --bun shadcn@latest add https://neu.beyondtheinnovation.com/r/neu-card.json
```

## Usage

```tsx
import {
  Card,
  CardContent,
  CardDescription,
  CardFooter,
  CardHeader,
  CardTitle,
} from "@/components/ui/neu/card"
```

```tsx
<Card>
  <CardHeader>
    <CardTitle>Card Title</CardTitle>
    <CardDescription>Card Description</CardDescription>
  </CardHeader>
  <CardContent>Card Content</CardContent>
  <CardFooter>Card Footer</CardFooter>
</Card>
```

## Size

`size="sm"` tightens the padding and gaps from 24px to 16px and drops the title to `text-sm`. Header, content and footer all read the same `--card-spacing` variable.

```tsx
<Card size="sm">
  <CardHeader>
    <CardTitle>Compact card</CardTitle>
  </CardHeader>
  <CardContent>Card Content</CardContent>
</Card>
```
