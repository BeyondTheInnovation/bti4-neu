## Installation

```bash
bunx --bun shadcn@latest add https://neu.beyondtheinnovation.com/r/neu-sonner.json
```

## Usage

```tsx
import { toast } from "sonner"

import { Button } from "@/components/ui/neu/button"
import { Toaster } from "@/components/ui/neu/sonner"
```

```tsx
// once, in your root layout
<Toaster />

// anywhere, with toast from "sonner"
<Button onClick={() => toast("Event has been created.")}>Show toast</Button>
```
